\begin{frame}
    \frametitle{Символ Лежандра. Повторення}
    \small
    \begin{block}{Означення}
        Нехай $p>2$ -- просте, $a\in\mathbb{Z}$. \textbf{Символ Лежандра}:
        \vspace{-1mm}
        \begin{equation*}
            \legendre{a}{p}=
            \begin{cases}
                \ \ 0, & p\mid a,\\
                \ \ 1, & a\in\mathcal{Q}_p\ \ (\exists x:\ x^2\equiv a \bmod p),\\
                -1, & a\in\overline{\mathcal{Q}}_p.
            \end{cases}
        \end{equation*}
    \end{block}
    \vspace{-2mm}
    \begin{exampleblock}{Закони}
        \begin{enumerate}[itemsep=2pt]
            \item $a\equiv b \pmod p\ \Rightarrow\ \legendre{a}{p}=\legendre{b}{p}$;
            \item $\legendre{ab}{p}=\legendre{a}{p}\legendre{b}{p}$, \ зокрема $\legendre{a^2}{p}=1$ при $p\nmid a$;
            \item критерій Ойлера: $\legendre{a}{p}\equiv a^{\frac{p-1}{2}} \pmod p$;
            \item $\legendre{1}{p}=1,\quad \legendre{-1}{p}=(-1)^{\frac{p-1}{2}},\quad \legendre{2}{p}=(-1)^{\frac{p^2-1}{8}}$;
            \item закон взаємності: для різних непарних простих $p,q$ \
            $\legendre{p}{q}\legendre{q}{p}=(-1)^{\frac{p-1}{2}\cdot\frac{q-1}{2}}$.
        \end{enumerate}
    \end{exampleblock}
\end{frame}

\begin{frame}
    \frametitle{Символ Якобі. Псевдоквадрати}
    \footnotesize
    \begin{block}{Означення}
        Нехай $n$ -- непарне, $n=p_1^{\alpha_1}\cdots p_k^{\alpha_k}$. \textbf{Символ Якобі}: \
        $\legendre{a}{n}=\legendre{a}{p_1}^{\alpha_1}\cdots\legendre{a}{p_k}^{\alpha_k}$.
    \end{block}
    \vspace{-2mm}
    \begin{exampleblock}{Закони \normalfont(для непарних $m,n$)}
        \begin{columns}[T]
            \begin{column}{0.47\textwidth}
                \begin{enumerate}[itemsep=1pt]
                    \item $a\equiv b \pmod n\Rightarrow\legendre{a}{n}=\legendre{b}{n}$;
                    \item $\legendre{ab}{n}=\legendre{a}{n}\legendre{b}{n}$;
                    \item $\legendre{a}{mn}=\legendre{a}{m}\legendre{a}{n}$;
                \end{enumerate}
            \end{column}
            \begin{column}{0.5\textwidth}
                \begin{enumerate}[start=4,itemsep=1pt]
                    \item $\legendre{-1}{n}=(-1)^{\frac{n-1}{2}}$, \ $\legendre{2}{n}=(-1)^{\frac{n^2-1}{8}}$;
                    \item $\gcd(m,n)=1$: \ $\legendre{m}{n}\legendre{n}{m}=(-1)^{\frac{m-1}{2}\cdot\frac{n-1}{2}}$.
                \end{enumerate}
            \end{column}
        \end{columns}
        Обчислюється за $O(\log^2 n)$ \textbf{без розкладу} $n$ на множники.
    \end{exampleblock}
    \vspace{-2mm}
    \begin{alertblock}{Псевдоквадрати}
        $\legendre{a}{n}=-1\Rightarrow a\notin\QR_n$, \ але \ $\legendre{a}{n}=1\not\Rightarrow a\in\QR_n$.

        \textbf{Означення.} $a\in\mathbb{Z}_n^*$ -- \textbf{псевдоквадрат} за модулем $n$, якщо $\legendre{a}{n}=1$, але $a\notin\QR_n$. Для $n=pq$:
        \vspace{-1mm}
        \begin{equation*}
            \PSQ_n=\left\{a\in\mathbb{Z}_n^*:\ \legendre{a}{p}=\legendre{a}{q}=-1\right\},\qquad |\QR_n|=|\PSQ_n|=\tfrac{\varphi(n)}{4}.
        \end{equation*}
        \vspace{-4mm}
    \end{alertblock}
\end{frame}

\begin{frame}
    \frametitle{Безпека DH}
    \small
    Нехай $G=\langle g\rangle$ -- циклічна група порядку $q$, $a,b,c\inR\mathbb{Z}_q$.
    \begin{block}{Задача Діффі-Геллмана (злам схеми DH)}
        Знаючи $(G,\,g,\,g^{a},\,g^{b})$ -- тобто все, що бачить Єва, -- знайти спільний ключ $g^{ab}$.
    \end{block}
    \vspace{-2mm}
    \begin{block}{CDH (Computational Diffie-Hellman)}
        Для будь-якого PPT-алгоритму $\mathcal{A}$: \
        $\Pr\left[\mathcal{A}(g,g^{a},g^{b})=g^{ab}\right]\le\mathsf{negl}(\lambda)$.
    \end{block}
    \vspace{-2mm}
    \begin{block}{DDH (Decisional Diffie-Hellman)}
        Набори $(g,g^{a},g^{b},g^{ab})$ та $(g,g^{a},g^{b},g^{c})$ обчислювально нерозрізненні:
        \vspace{-1mm}
        \begin{equation*}
            \left|\Pr\left[\mathcal{A}(g,g^{a},g^{b},g^{ab})=1\right]-\Pr\left[\mathcal{A}(g,g^{a},g^{b},g^{c})=1\right]\right|\le\mathsf{negl}(\lambda).
        \end{equation*}
    \end{block}
    \vspace{-2mm}
    \begin{alertblock}{}
        \footnotesize
        Розв'язуємо DLP $\Rightarrow$ розв'язуємо CDH $\Rightarrow$ розв'язуємо DDH. Отже, припущення DDH -- найсильніше з трьох.
        У всій групі $\mathbb{Z}_p^*$ DDH \textbf{не виконується}: $\legendre{g^{x}}{p}=(-1)^{x}$ видає парність $x$.
    \end{alertblock}
\end{frame}

\section{Схема Горнера}
%------------------------------------------------
\begin{frame}
    \frametitle{Схема}
    \small
    \textbf{Вхід:} $a,m\in\mathbb{N}\ (a<m)$, \ $n$, \ $|n|=k$ біт. \quad \textbf{Обчислюємо:} $a^{n}\bmod m$.
    \begin{block}{Ідея}
        \vspace{-4mm}
        \begin{gather*}
            n=(n_{k-1}n_{k-2}\ldots n_1n_0)_2=\textstyle\sum_{i=0}^{k-1}n_i2^{i}\\
            a^{n}=a^{\sum_{i=0}^{k-1}n_i2^{i}}=\Big(\ldots\big(\big(a^{n_{k-1}}\big)^{2}\cdot a^{n_{k-2}}\big)^{2}\cdot a^{n_{k-3}}\ldots\Big)^{2}\cdot a^{n_0}
        \end{gather*}
        \vspace{-5mm}

        {\footnotesize Після кожного піднесення до квадрату та множення беремо число за $\bmod\ m$.}
    \end{block}
    \vspace{-2mm}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{exampleblock}{Ітеративно}
                \begin{enumerate}
                    \item $b:=1$
                    \item \textbf{for} $i=k-1$ \textbf{downto} $0$:\\
                          \quad $b:=b^{2}\bmod m$\\
                          \quad \textbf{if} $n_i=1$: \ $b:=b\cdot a\bmod m$
                    \item \textbf{return} $b$
                \end{enumerate}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.44\textwidth}
            \begin{alertblock}{Складність}
                $k$ піднесень до квадрату $+$ $\mathrm{wt}(n)\le k$ множень: \ $\le 2k$ модулярних множень замість $n-1$.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}
